DDL

CREATE TABLE producao.apontamento (
id bigint GENERATED ALWAYS AS IDENTITY,
ordem_id bigint NOT NULL,
maquina_id bigint NOT NULL,
operador_id bigint NOT NULL,
quantidade numeric(12,3) NOT NULL,
refugo numeric(12,3) NOT NULL DEFAULT 0,
registrado_em timestamptz NOT NULL DEFAULT now(),
CONSTRAINT pk_apontamento PRIMARY KEY (id),
CONSTRAINT fk_apontamento_ordem FOREIGN KEY (ordem_id) REFERENCES producao.ordem_producao (id),
CONSTRAINT fk_apontamento_maquina FOREIGN KEY (maquina_id) REFERENCES producao.maquina (id),
CONSTRAINT fk_apontamento_operador FOREIGN KEY (operador_id) REFERENCES producao.operador (id)
);
CREATE INDEX ix_apontamento_ordem ON producao.apontamento (ordem_id);
CREATE INDEX ix_apontamento_maquina_data ON producao.apontamento (maquina_id, registrado_em);
